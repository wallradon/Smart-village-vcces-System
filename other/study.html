1 PART 1: จาก Mock Data → API (ฝั่ง Frontend)

1.1 มุมมองของ Frontend

💡 หลักคิดสำคัญ: ฝั่ง Frontend ไม่ต้องรู้ว่าข้อมูลมาจากไฟล์ JSON ในเครื่อง หรือจาก Server ที่อยู่ไกล หน้าที่ของเราคือ "ส่งคำขอ (Request) แล้วรอผลตอบกลับ (Response) มาแสดงผล" การเขียนให้เป็นระเบียบจะทำให้ทีหลังเปลี่ยนแหล่งข้อมูลได้โดยไม่ต้องแก้หน้า HTML เลย

1.2 ขั้นตอนที่ 1 — แยก Data Layer ออกเป็นไฟล์ api.js

admin/api.js — สร้างไฟล์ใหม่ไว้จัดการข้อมูลทั้งหมด (Separation of Concerns)
// ไฟล์นี้มีหน้าที่เดียว: รับส่งข้อมูลกับแหล่งข้อมูล (Data Layer)
// โค้ดหน้า UI จะเรียกใช้แค่ฟังก์ชันพวกนี้ ไม่ต้องรู้ที่มาของข้อมูล

const API = {

    // เรียกข้อมูลลูกบ้านทั้งหมด
    async getUsers() {
        const res = await fetch("./dataTest.json");
        if (!res.ok) throw new Error(`HTTP error: ${res.status}`);
        return res.json();
    },

    // เรียกข้อมูลลูกบ้านคนเดียว
    async getUser(id) {
        const res = await fetch(`./dataTest.json`);
        if (!res.ok) throw new Error(`HTTP error: ${res.status}`);
        const users = await res.json();
        return users.find(u => u.id === Number(id));
    }
};
💡 ประโยชน์: ต่อไปถ้ามี Server จริง แค่เปลี่ยนบรรทัด URL ในไฟล์ api.js ที่เดียว ส่วน script.js (การแสดงผล) ไม่ต้องแตะเลย

1.3 ขั้นตอนที่ 2 — จำลอง API ด้วย Promise (เมื่อยังไม่มี Server)

admin/api.js — จำลองความล่าช้าของเครือข่าย (ดูเหมือน API จริง)
const API = {

    async getUsers() {
        // จำลองว่า Server ใช้เวลาตอบ 1.5 วิ (ตอน dev จะได้เห็น loading state)
        await new Promise(resolve => setTimeout(resolve, 1500));
        const res = await fetch("./dataTest.json");
        if (!res.ok) throw new Error(`HTTP error: ${res.status}`);
        return res.json();
    },

    // จำลอง error (เพื่อทดสอบหน้าแสดงข้อผิดพลาด)
    async getUsersFail() {
        await new Promise(resolve => setTimeout(resolve, 1500));
        throw new Error("Network error");   // ปล่อยให้ฝั่ง caller จัดการ
    }
};

1.4 ขั้นตอนที่ 3 — ใช้ Async/Await + try/catch ในหน้าแสดงผล

admin/script.js — ดึงข้อมูลผ่าน Data Layer (โค้ดสะอาดกว่าเดิม)
async function load() {
    setLoading();   // 1. แสดงสถานะ "กำลังโหลด"
    try {
        mainData = await API.getUsers();   // 2. ขอข้อมูล
        renderUserList(mainData);            // 3. วาดข้อมูลลงหน้า
    } catch (err) {
        showError(err.message);              // 4. มีปัญหา → แสดง error
    }
}

1.5 ขั้นตอนที่ 4 — จัดการ 3 สถานะ: Loading / Error / Success

Loading
แสดงข้อความ/ตัวหมุน (spinner) ขณะรอ
→
Error
แสดงข้อความแจ้ง + เปิดปุ่มลองใหม่
→
Success
วาดข้อมูลจริงบนหน้า
admin/script.js — ฟังก์ชันจัดการสถานะ
function setLoading() {
    UserData.innerHTML = `<p class="loading-text">⏳ Loading data...</p>`;
}

function showError(msg) {
    UserData.innerHTML = `<p class="loading-text" style="color:red">
        ❌ ${msg} — <button onclick="load()">ลองใหม่</button></p>`;
}

1.6 เทคนิคขั้นสูง — ยกเลิก request ที่ไม่ต้องการ (AbortController)

admin/api.js — ยกเลิก fetch ถ้าผู้ใช้เปลี่ยนหน้าเร็วเกินไป
let currentAbort = null;

async getUsers() {
    // ยกเลิก request ก่อนหน้า (ถ้ายังไม่เสร็จ) กันข้อมูลมาซ้อนกัน
    if (currentAbort) currentAbort.abort();
    currentAbort = new AbortController();

    const res = await fetch("./dataTest.json", { signal: currentAbort.signal });
    if (!res.ok) throw new Error(`HTTP error: ${res.status}`);
    return res.json();
}

1.7 สรุปเส้นทางข้อมูลใหม่

ชั้น (Layer) ไฟล์ หน้าที่
Data Layer api.js (ใหม่) จัดการ fetch / จำลอง API / ตรวจ error
Logic Layer script.js เรียกใช้ api.js และวาดผล (render)
View Layer admin.html โครงหน้าเว็บ ไม่ยุ่งกับข้อมูล

2 PART 2: Login แบบ Professional (ฝั่ง Frontend)

2.1 มองจากฝั่ง Frontend — ปัญหาของ login.js เดิม

⚠️ ปัญหาที่เห็นได้ชัดจากฝั่ง Frontend:
  • รหัสผ่าน admin/123 เปิดในโค้ดใครก็เห็น — อย่างน้อยควร "ปิด" การทำงานนี้ไว้เบื้องหลัง
  • กดปุ่ม Enter หรือ Submit ซ้ำได้เรื่อย ๆ → ส่ง request ซ้ำ
  • ไม่มี Validation — กรอกอะไรก็ส่งไปหมด
  • ไม่มีสถานะ "กำลังตรวจสอบ" — ผู้ใช้ไม่รู้ว่าระบบทำงานอยู่หรือค้าง
  • หน้า Admin กันแค่ sessionStorage ที่แก้กันง่าย

Frontend ทำได้เต็มที่ 3 อย่าง: UX (หน้าตา/การแจ้งเตือน), Validation (ตรวจความถูกต้อง), และ Route Guard (ป้องกันหน้า)

2.2 Client-side Validation — เช็คก่อนส่ง

login/login.js — ตรวจว่ากรอกครบ + รูปแบบถูกก่อนส่ง
function validate(username, password) {
    if (!username.trim()) return "กรุณากรอกชื่อผู้ใช้";
    if (!password)       return "กรุณากรอกรหัสผ่าน";
    if (password.length < 3) return "รหัสผ่านสั้นเกินไป";
    return null;   // ผ่านทั้งหมด → คืนค่า null (ไม่มีข้อผิดพลาด)
}

// ใช้ใน submit handler
const err = validate(inputUsername, inputPassword);
if (err) {
    showError(err);
    return;   // หยุด ไม่ส่ง request
}
💡 หมายเหตุ: Validation ฝั่ง Frontend เป็นแค่ "ชั้นแรก" เพื่อ UX ที่ดี ปลอดภัยจริงต้องตรวจซ้ำที่ Server ด้วย — แต่สำหรับเรียนจบ Frontend คือสิ่งที่เราทำได้

2.3 UI States — Loading / Error / Success

login/login.js — กันกดซ้ำ + แสดงสถานะทุกช่วง
loginForm.addEventListener("submit", async (event) => {
    event.preventDefault();

    // ---- Loading state: ปิดปุ่มกันกดซ้ำ + เปลี่ยนข้อความ
    submitBtn.disabled = true;
    submitBtn.textContent = "⏳ กำลังตรวจสอบ...";
    hideError();

    try {
        // จำลองการตรวจสอบ (Delay) — ต่อไปค่อยส่งไป Server ตรงนี้
        await new Promise(resolve => setTimeout(resolve, 1000));

        if (inputUsername === "admin" && inputPassword === "123") {
            // ---- Success: ออกบัตรผ่าน แล้วไปหน้า Admin
            sessionStorage.setItem("isLoggedIn", "true");
            window.location.href = "../admin/admin.html";
        } else {
            // ---- Error: แจ้งให้รู้
            showError("ชื่อผู้ใช้หรือรหัสผ่านไม่ถูกต้อง");
        }
    } catch (err) {
        showError("เกิดข้อผิดพลาด กรุณาลองใหม่");
    } finally {
        // คืนปุ่มเป็นปกติเสมอ (ทั้งสำเร็จและพลาด)
        submitBtn.disabled = false;
        submitBtn.textContent = "เข้าสู่ระบบ";
    }
});

2.5 จัดการ Session อย่างถูกวิธี

ที่เก็บ อยู่ได้นานแค่ไหน เหมาะกับ
sessionStorage ปิดแท็บ = หาย บัตรผ่าน "ชั่วคราว" ของระบบนี้ ✅
localStorage อยู่จนกว่าจะลบเอง ของที่จำข้ามครั้ง เช่น ธีม, token
ตัวแปรใน JS รีเฟรชหน้า = หาย ค่าชั่วคราวที่ไม่ต้องเก็บ
💡 ปกติระบบ Login ควรเก็บ "token" ที่ Server ออกให้ แต่ถ้าขอบเขตเรียนแค่ Frontend วิธีที่ยอมรับได้คือ เก็บค่าสถานะใน sessionStorage แบบที่โปรเจกต์ทำ และปรับให้สมบูรณ์ตามข้อ 2.6–2.8

2.6 Route Guard ฝั่งหน้าเว็บ — กันเปิดหน้า Admin ตรง ๆ

admin/script.js — บรรทัดแรกสุดของไฟล์ (เดิมมีแล้ว ทำให้แน่นขึ้น)
// 1. ยังไม่ล็อกอิน → เด้งกลับหน้า Login
if (sessionStorage.getItem("isLoggedIn") !== "true") {
    // จำว่าตอนแรกอยากเข้าหน้าไหน (ต่อไปจะได้กลับมาที่เดิมได้)
    sessionStorage.setItem("redirectAfterLogin", location.href);
    window.location.href = "../login/login.html";
    throw new Error("No permission");   // หยุดรันโค้ดที่เหลือทันที
}

2.7 Auto-logout เมื่อไม่ขยับนาน (Session Timeout)

admin/script.js — กันคนลืมล็อกเอาต์ทิ้งเครื่อง
// หมดเวลา 5 นาที → ออกจากระบบอัตโนมัติ
const TIMEOUT = 5 * 60 * 1000;
let timer;

function resetTimer() {
    clearTimeout(timer);
    timer = setTimeout(logout, TIMEOUT);   // ไม่ขยับครบเวลา → logout
}

// ทุกครั้งที่ผู้ใช้ขยับเมาส์/พิมพ์ → นับเวลาตั้งแต่ต้น
['mousemove', 'keydown', 'click'].forEach(evt =>
    document.addEventListener(evt, resetTimer)
);
resetTimer();

2.8 Logout ที่ล้างข้อมูลครบถ้วน

admin/script.js — ล้างทุกอย่างที่เกี่ยวข้องกับ session
function logout() {
    sessionStorage.removeItem("isLoggedIn");         // ลบบัตรผ่าน
    sessionStorage.removeItem("redirectAfterLogin");  // ลบที่อยู่ที่จำไว้
    window.location.href = "../index.html";             // กลับหน้าแรก
}

3 PART 3: การออกแบบและสร้าง Admin System (Admin System Architecture)

3.1 โครงสร้างสถาปัตยกรรมระบบ Admin (System Architecture)

💡 หลักคิดระบบ Admin: ระบบผู้ดูแลระบบที่มีประสิทธิภาพต้องเป็น Single Page Application (SPA) คือผู้ใช้เปลี่ยนหน้า (Home / Vehicle / User Data) โดยไม่ต้องโหลดเว็บใหม่ ทำให้การใช้งานลื่นไหล ไม่กระตุก

3.2 ระบบ Routing สลับหน้าจอ (SPA Navigation)

admin/script.js — สลับหน้าจอโดยการเปลี่ยน class 'active'
function showPage(target, params) {
    // 1. ซ่อนทุกหน้า และเอาไฮไลต์เมนูกลับเป็นปกติก่อน
    document.querySelectorAll('.page').forEach(page => page.classList.remove('active'));
    document.querySelectorAll('nav li').forEach(li => li.classList.remove('user-select'));

    // 2. แสดงเฉพาะหน้าเป้าหมาย
    const targetPage = document.querySelector(`#page-${target}`);
    if (targetPage) {
        targetPage.classList.add('active');
        renderUserPage(target, params); // เรียกฟังก์ชันวาดข้อมูลของหน้านั้น
    }

    // 3. ไฮไลต์เมนูที่เลือก
    const activeLi = document.querySelector(`nav li[data-target="${target}"]`);
    if (activeLi) activeLi.classList.add('user-select');
}

3.3 การดึงและวนลูปแสดงข้อมูลแบบ Nested Data (Relational Rendering)

admin/script.js — แตกรายการเข้า-ออกของยานพาหนะทุกคัน (Nested Loops)
function renderVehicleList(users) {
    const container = document.querySelector('#VehicleData');
    let html = "";

    users.forEach(user => {
        if (user.vehicles && user.vehicles.length > 0) {
            user.vehicles.forEach(vehicle => {
                // วนลูปอ่านประวัติการเข้า-ออกย่อยทุกรอบ
                if (vehicle.timeInOut && vehicle.timeInOut.length > 0) {
                    vehicle.timeInOut.forEach(record => {
                        html += `
                        <div class="User VehicleRow">
                            <h2>${vehicle.type}</h2>
                            <h2>${vehicle.plate}</h2>
                            <h2>in ${record.in} out ${record.out}</h2>
                        </div>`;
                    });
                }
            });
        }
    });
    container.innerHTML = html;
}

3.4 การเตรียมความพร้อมรองรับ CRUD กับ Backend ในอนาคต

การทำงาน (CRUD) HTTP Method หน้าที่ในฝั่ง Frontend Admin
Create (สร้าง) POST ส่งข้อมูลยานพาหนะ / ลูกบ้านใหม่ ไปบันทึกในระบบ
Read (อ่าน) GET ดึงรายการลูกบ้าน และเวลาเข้า-ออกรถยนต์มาแสดง
Update (แก้ไข) PUT / PATCH อัปเดตข้อมูลบ้านเลขที่ ป้ายทะเบียน หรือสถานะสมาชิก
Delete (ลบ) DELETE ยกเลิกข้อมูลลูกบ้าน หรือลบป้ายทะเบียนออกจากระบบ
