1.3 โค้ด Express API Server ฉบับสมบูรณ์ (Complete server.js)

โค้ดชุดนี้รวมการอ่านไฟล์ JSON, Middleware, Error Handling, CRUD Endpoints และ Authentication ไว้ครบถ้วน:

server.js — ฉบับเต็ม (มีระบบอ่าน-เขียนไฟล์ JSON และจัดการ Error ครอบคลุม)
const express = require("express");
const cors = require("cors");
const fs = require("fs");
const path = require("path");

const app = express();
const PORT = process.env.PORT || 3000;
const DATA_FILE = path.join(__dirname, "admin", "dataTest.json");

// ==================== 1. MIDDLEWARES ====================
// อนุญาตให้เว็บจาก Origin อื่น (เช่น Frontend) ดึง API ได้
app.use(cors());

// แปลง JSON Body จาก Request (req.body) ให้อยู่ในรูปแบบ JavaScript Object
app.use(express.json());

// Custom Logging Middleware สำหรับบันทึกการเรียกใช้งาน API ทุกครั้ง
app.use((req, res, next) => {
    console.log(`[${new Date().toISOString()}] ${req.method} ${req.url}`);
    next();
});

// ==================== 2. HELPER FUNCTIONS ====================
// อ่านข้อมูลจากไฟล์ dataTest.json
function readData() {
    try {
        const raw = fs.readFileSync(DATA_FILE, "utf8");
        return JSON.parse(raw);
    } catch (err) {
        console.error("Error reading JSON file:", err);
        return [];
    }
}

// บันทึกข้อมูลลงไฟล์ dataTest.json
function saveData(data) {
    try {
        fs.writeFileSync(DATA_FILE, JSON.stringify(data, null, 2), "utf8");
        return true;
    } catch (err) {
        console.error("Error writing JSON file:", err);
        return false;
    }
}

// ==================== 3. REST API ENDPOINTS ====================

/**
 * @route   GET /api/residents
 * @desc    ดึงรายชื่อและข้อมูลลูกบ้านทั้งหมด
 */
app.get("/api/residents", (req, res) => {
    const residents = readData();
    res.status(200).json({
        success: true,
        count: residents.length,
        data: residents
    });
});

/**
 * @route   GET /api/residents/:id
 * @desc    ดึงข้อมูลลูกบ้านรายคนตาม ID
 */
app.get("/api/residents/:id", (req, res) => {
    const residents = readData();
    const resident = residents.find(u => u.id === Number(req.params.id));

    if (!resident) {
        return res.status(404).json({
            success: false,
            message: `ไม่พบข้อมูลลูกบ้านรหัส ID ${req.params.id}`
        });
    }

    res.status(200).json({ success: true, data: resident });
});

/**
 * @route   GET /api/vehicles
 * @desc    ดึงรายการประวัติการเข้า-ออกของยานพาหนะทุกคัน
 */
app.get("/api/vehicles", (req, res) => {
    const residents = readData();
    let vehicleLogs = [];

    residents.forEach(r => {
        if (r.vehicles && Array.isArray(r.vehicles)) {
            r.vehicles.forEach(v => {
                vehicleLogs.push({
                    residentId: r.id,
                    houseNumber: r.houseNumber,
                    ownerName: r.ownerName,
                    plate: v.plate,
                    type: v.type,
                    timeInOut: v.timeInOut || []
                });
            });
        }
    });

    res.status(200).json({
        success: true,
        totalVehicles: vehicleLogs.length,
        data: vehicleLogs
    });
});

/**
 * @route   POST /api/residents
 * @desc    เพิ่มข้อมูลลูกบ้านคนใหม่
 */
app.post("/api/residents", (req, res) => {
    const { houseNumber, ownerName, role } = req.body;

    if (!houseNumber || !ownerName) {
        return res.status(400).json({
            success: false,
            message: "กรุณากรอก houseNumber และ ownerName ให้ครบถ้วน"
        });
    }

    const residents = readData();
    const newId = residents.length > 0 ? Math.max(...residents.map(r => r.id)) + 1 : 1;

    const newResident = {
        id: newId,
        houseNumber,
        ownerName,
        role: role || "resident",
        regitter: new Date().toLocaleDateString("th-TH"),
        dateMember: new Date().toLocaleDateString("th-TH"),
        MemberTimeout: "31/12/2026",
        vehicles: []
    };

    residents.push(newResident);
    if (saveData(residents)) {
        res.status(201).json({ success: true, message: "เพิ่มลูกบ้านสำเร็จ", data: newResident });
    } else {
        res.status(500).json({ success: false, message: "ไม่สามารถบันทึกข้อมูลได้" });
    }
});

/**
 * @route   POST /api/login
 * @desc    ระบบเข้าสู่ระบบและออก Token ชั่วคราว
 */
app.post("/api/login", (req, res) => {
    const { username, password } = req.body;
    const residents = readData();

    const user = residents.find(u => u.username === username && u.password === password);

    if (user || (username === "admin" && password === "123")) {
        res.status(200).json({
            success: true,
            message: "เข้าสู่ระบบสำเร็จ",
            token: "bearer-vcces-mock-token-xyz123",
            user: user ? { id: user.id, houseNumber: user.houseNumber, ownerName: user.ownerName, role: user.role } : { ownerName: "ผู้ดูแลระบบ", role: "admin" }
        });
    } else {
        res.status(401).json({ success: false, message: "ชื่อผู้ใช้หรือรหัสผ่านไม่ถูกต้อง" });
    }
});

// ==================== 4. 404 & GLOBAL ERROR HANDLER ====================
// ดักจับ Route ที่ไม่มีอยู่จริง
app.use((req, res) => {
    res.status(404).json({ success: false, message: "Endpoint Not Found" });
});

// ดักจับ Internal Server Error ทั่วไป
app.use((err, req, res, next) => {
    console.error(err.stack);
    res.status(500).json({ success: false, message: "Internal Server Error" });
});

// เปิดรันเซิร์ฟเวอร์
app.listen(PORT, () => {
    console.log(`==================================================`);
    console.log(`🚀 Express API Server is running on port ${PORT}`);
    console.log(`🔗 URL: http://localhost:${PORT}/api/residents`);
    console.log(`==================================================`);
});

1.4 สรุป HTTP Status Codes ที่ต้องรู้สำหรับการสร้าง API

Status Code ประเภท / ความหมาย สถานการณ์ที่ใช้ใน Express Server
200 OK สำเร็จ (Success) ดึงข้อมูล GET หรือค้นหาข้อมูลเจอสำเร็จ
201 Created สร้างสำเร็จ (Created) บันทึกเพิ่มข้อมูลลูกบ้านคนใหม่สำเร็จด้วย POST
400 Bad Request คำขอไม่ถูกต้อง ผู้ใช้ลืมกรอกข้อมูลที่จำเป็นมาใน Body (req.body)
401 Unauthorized ไม่ผ่านการรับรองสิทธิ์ กรอก Username/Password ล็อกอินไม่ถูกต้อง
404 Not Found ไม่พบทรัพยากร ค้นหา ID ลูกบ้านไม่พบ หรือกรอก URL API ผิด
500 Internal Error เซิร์ฟเวอร์ผิดพลาด เกิดข้อผิดพลาดในการอ่านไฟล์ dataTest.json หรือโค้ดพัง

2 API Documentation Specification (เอกสารข้อตกลงและมาตรฐาน API)

📄 วัตถุประสงค์ของเอกสารนี้: ใช้สำหรับส่งต่อให้โปรแกรมเมอร์ฝั่ง Backend / Node.js Express เพื่อเป็น API Contract ข้อตกลงมาตรฐานโครงสร้าง Response และ HTTP Endpoints ให้ส่งข้อมูลกลับมาตรงกับที่ฝั่ง Frontend ของเราออกแบบไว้

2.1 มาตรฐานโครงสร้างข้อมูลตอบกลับ (Standard Response Structure)

กำหนดให้ทุก Endpoint ตอบกลับในรูปแบบ JSON Object ที่มี Key หลักดังนี้เสมอ:

JSON — Standard Success & Error Response Format
// 1. กรณีทำงานสำเร็จ (Success Response)
{
  "success": true,
  "message": "ข้อความอธิบายผลลัพธ์ (Optional)",
  "count": 3,             // จำนวนรายการ (ถ้าเป็น Array)
  "data": [ ... ]         // ข้อมูลผลลัพธ์ (Object หรือ Array)
}

// 2. กรณีเกิดข้อผิดพลาด (Error Response)
{
  "success": false,
  "message": "สาเหตุของข้อผิดพลาด เช่น ไม่พบข้อมูล หรือรหัสผ่านไม่ถูกต้อง",
  "errorCode": "RESIDENT_NOT_FOUND" // รหัสอ้างอิง error (Optional)
}

2.2 รายละเอียด API Endpoints ทั้งหมด (API Reference Specification)

GET /api/residents

ดึงรายชื่อและข้อมูลลูกบ้านทั้งหมด

Description: ส่งคืนข้อมูลลูกบ้าน รายละเอียดบ้านเลขที่ ทะเบียนรถ และประวัติเข้า-ออกทั้งหมด

Headers Required: Content-Type: application/json

Success Response Example (200 OK):

{
  "success": true,
  "count": 1,
  "data": [
    {
      "id": 1,
      "houseNumber": "158/1",
      "ownerName": "สมหมาย ดีใจ",
      "role": "resident",
      "regitter": "15/10/2026",
      "dateMember": "15/10/2026",
      "MemberTimeout": "15/11/2026",
      "vehicles": [
        {
          "plate": "1กข1111",
          "type": "Car",
          "regitter": "15/10/2026",
          "timeInOut": [{ "in": "15:00", "out": "19:00" }]
        }
      ]
    }
  ]
}

GET /api/residents/:id

ดึงข้อมูลลูกบ้านระบุตาม ID

URL Parameter: id (Number) เช่น /api/residents/1

Success Response (200 OK): คืนค่า Object ลูกบ้าน 1 รายการใน Key data

Error Response (404 Not Found):

{
  "success": false,
  "message": "ไม่พบข้อมูลลูกบ้านรหัส ID 99"
}

GET /api/vehicles

ดึงรายการประวัติการเข้า-ออกของยานพาหนะทั้งหมด

Success Response Example (200 OK):

{
  "success": true,
  "totalVehicles": 2,
  "data": [

      "houseNumber": "158/1",
